app.css
/* fitted to the reference over 19
   frames — max error .007 of the
   travel. easeInOutQuad misses by
   .071, Material's by .275 */
--ease-open: cubic-bezier(
  .55, .08, .4, .95);

/* read off the WIDTH: at 5:1 a
   picked perspective fans the far
   end of the pill */
.scry__seat { perspective:
  calc(var(--w) * 2.2); }

/* the panel is hinged on its own
   long centre line — the line the
   icon already sits on */
.scry__field {
  transform-origin: 50% 50%;
  transition: transform .633s
    var(--ease-open);
}
[data-state="closed"] .scry__field {
  transform: rotateX(-90deg);
}
app.js
// one button, three jobs, in this
// order — a control that opens,
// clears and closes under ONE label
// is three controls in a trench coat
btn.addEventListener('click', () => {
  if (!isOpen()) { open(); return; }
  if (hasText()) { clear(); return; }
  close({ focus: true });
});

const name = () =>
  !isOpen() ? 'Search'
  : hasText() ? 'Clear search'
  : 'Close search';

// :focus-visible ALWAYS matches a
// focused text field, so a ring
// gated on it shows for every mouse
// user. Track the modality by hand.
addEventListener('keydown', (e) => {
  if (KEYS.has(e.key)) kbd = true;
}, true);
